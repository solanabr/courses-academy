fn try_accounts(program_id, views, ix_data, bumps) -> Result<Self> {
    // PHASE 1 + 2, field by field, in declaration order:
    let payer: Signer = anchor::try_load(views)?;          // load: signature bit
    anchor::constraint_mut(&payer)?;                       // constraint: mut

    let marquee: Account<Marquee> = anchor::try_init(       // load + init CPI
        views, payer, program_id)?;                        //   writes account:Marquee disc

    let system_program: Program<System> =
        anchor::try_load(views)?;                          // load: address + executable

    Ok(LightMarquee { payer, marquee, system_program })     // PHASE 3 handoff
}
